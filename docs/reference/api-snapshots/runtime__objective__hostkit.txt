package hostkit // import "github.com/wsnacj/agentx-go/runtime/objective/hostkit"

Package hostkit composes the canonical Objective ingress, host-owned runtime
adapter dispatch, observation normalization, and verification mechanisms.

The package is part of the v0.2.2 Developer Preview compatibility surface.
It does not provide a provider, Runner, backend, credential, product policy,
or production side effect.

TYPES

type Config struct {
	Handler  Handler
	Handlers map[objective.DisplaySafeRef]Handler
}
    Config contains only host-owned adapter capabilities. It does not install a
    provider, backend, credential, approval, policy, or product default.

type DispatchInput struct {
	Enabled                  bool
	HostConfirmed            bool
	Request                  objective.RuntimeAdapterRequest
	Handler                  Handler
	Handlers                 map[objective.DisplaySafeRef]Handler
	ExpectedObservationKinds []string
	Boundaries               []objective.Boundary
	Context                  context.Context
}
    DispatchInput controls one explicitly confirmed host dispatch.

func Dispatch(input DispatchInput) DispatchResult
    Dispatch invokes exactly one host handler after request and confirmation
    gates, then runs canonical observation normalization and verification.

func (report DispatchResult) Normalize() DispatchResult
    Normalize returns a defensive, canonical view of the dispatch result.

type Handler func(context.Context, objective.RuntimeAdapterRequest) objective.RuntimeAdapterResult
    Handler is a host-owned runtime-adapter execution function.

type RunRequest struct {
	Ingress               objective.ManagedObjectiveIngressInput
	DispatchEnabled       bool
	DispatchHostConfirmed bool
	Boundaries            []objective.Boundary
}
    RunRequest contains display-safe Objective inputs and explicit dispatch
    authorization for one Host Kit invocation.

type Runtime struct {
	// Has unexported fields.
}
    Runtime is an immutable Objective composition. Handler concurrency safety
    remains the responsibility of the Host.

func New(config Config) (*Runtime, error)
    New validates the host-owned adapter capabilities and creates an immutable
    Objective Runtime.

func (runtime *Runtime) Run(ctx context.Context, request RunRequest) RunResult
    Run projects one managed Objective and, only after explicit enable and host
    confirmation, invokes exactly one selected handler before verification.
